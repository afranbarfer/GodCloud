! --- Entrar en modo Admin ---
Switch-S1> enable
! --- Entrar en la VLAN database ---
Switch-S1# vlan database
! --- Añadir las vlans en la database---
Switch-S1(vlan)# vlan 10 name SERVICIOS
Switch-S1(vlan)# vlan 99 name GESTION
Switch-S1(vlan)# exit
Switch-S1# configure terminal
! --- Trunk hacia S4 ---
Switch-S1(config)# interface f0/0
Switch-S1(config-if)# description Trunk_a_S4
Switch-S1(config-if)# switchport mode trunk
Switch-S1(config-if)# switchport trunk allowed vlan 1,10,99,1002-1005
Switch-S1(config-if)# no shutdown

! --- Trunk hacia S5 ---
Switch-S1(config)# interface f0/1
Switch-S1(config-if)# description Trunk_a_S5
Switch-S1(config-if)# switchport mode trunk
Switch-S1(config-if)# switchport trunk allowed vlan 1,10,99,1002-1005
Switch-S1(config-if)# no shutdown

! --- Interfaces de Servicios ---
Switch-S1(config)# interface range f0/2 - 15
Switch-S1(config-if-range)# switchport mode access
Switch-S1(config-if-range)# switchport access vlan 10
Switch-S1(config-if-range)# no shutdown

! --- IP de Gestión ---
Switch-S1(config)# interface vlan 99
Switch-S1(config-if)# ip address 192.168.99.11 255.255.255.224
Switch-S1(config-if)# no shutdown

! --- Guardar Configuración ---
Switch-S1# copy running-config startup-config
