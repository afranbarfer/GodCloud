! --- Entrar en modo Admin ---
Switch-S5> enable
! --- Entrar en la VLAN database ---
Switch-S5# vlan database
! --- Añadir las vlans en la database---
Switch-S5(vlan)# vlan 10 name SERVICIOS
Switch-S5(vlan)# vlan 99 name GESTION
Switch-S5(vlan)# exit
Switch-S5# configure terminal

Switch-S5(config)# interface f0/0
Switch-S5(config-if)# description Trunk_a_S1
Switch-S5(config-if)# switchport mode trunk
Switch-S5(config-if)# switchport trunk allowed vlan 1,10,99,1002-1005
Switch-S5(config-if)# no shutdown
